# Hacker's Delight-style routines, written by hand the long way in the loop's term language (docs/corpora/hd-own/SOURCE.md)
isolate lowest 1	and(x, sub(0, x))
clear lowest 1	and(x, add(x, -1))
mask trailing 0s	and(xor(x, -1), add(x, -1))
mask through lowest 1	xor(x, add(x, -1))
propagate lowest 1 right	or(x, add(x, -1))
set lowest 0	or(x, add(x, sub(0, -1)))
isolate lowest 0	and(xor(x, -1), add(x, sub(0, -1)))
clear trailing 1s	and(x, add(x, sub(0, -1)))
floor average	add(and(x, y), shr1(xor(x, y), 0))
ceiling average	sub(or(x, y), shr1(xor(x, y), 0))
min via lt mask	xor(y, and(xor(x, y), sub(0, lt(x, y))))
max via lt mask	xor(x, and(xor(x, y), sub(0, lt(x, y))))
add via xor/and	add(xor(x, y), shl1(and(x, y), 0))
add via or/and	add(or(x, y), and(x, y))
sub via and/not	sub(and(x, xor(y, -1)), and(xor(x, -1), y))
xor via or-and	sub(or(x, y), and(x, y))
and via add-or	sub(add(x, y), or(x, y))
negate via not+1	add(xor(x, -1), sub(0, -1))
not via sub	sub(-1, x)
or via add	add(x, and(xor(x, -1), y))
or via de Morgan	xor(and(xor(x, -1), xor(y, -1)), -1)
bit select	xor(y, and(xor(x, y), z))
majority	or(or(and(x, y), and(x, z)), and(y, z))
swap-xor leftover	xor(xor(x, y), y)
3-way add	add(add(x, y), z)
